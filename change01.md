
这是一个架构上的重大变更。原本你的系统是一个 动态网站（每次请求查询数据库并渲染），现在你希望它变成一个 静态站点生成器 (SSG) 的 Web 管理版。
这意味着流程变为：
写作：用户在 Web 编辑器写作 -> 存入 SQLite。
构建：后端触发 mdbook 编译 -> 从数据库读取内容 -> 生成文件结构 -> 调用 mdbook 构建 -> 生成静态 HTML。
服务：Axum 直接托管生成的静态目录 (book/)，不再使用 Tera 动态渲染文章详情。

2. 创建构建逻辑模块 (src/book_gen.rs)
我们需要一个模块，负责把数据库里的文章“倒腾”出来，变成 mdbook 需要的文件结构 (SUMMARY.md 和各个 .md 文件)，然后执行构建。
新建 src/book_gen.rs：
code
Rust
// src/book_gen.rs
use crate::models::post::Post;
use anyhow::Result;
use mdbook::MDBook;
use rbatis::Rbatis;
use std::fs;
use std::path::Path;

// 定义 mdbook 的工作目录
const BOOK_ROOT: &str = "data/book_workspace";

pub async fn rebuild_book(rb: &Rbatis) -> Result<()> {
    println!("🔄 Starting mdBook build process...");

    // 1. 准备目录结构
    // data/book_workspace/
    // ├── book.toml
    // └── src/
    //     ├── SUMMARY.md
    //     └── ...posts...
    let src_dir = Path::new(BOOK_ROOT).join("src");
    if src_dir.exists() {
        fs::remove_dir_all(&src_dir)?;
    }
    fs::create_dir_all(&src_dir)?;

    // 2. 获取所有文章 (按时间倒序)
    // 注意：这里需要你确保 models/post.rs 里有按时间排序的查询，或者直接 select_all 拿到内存排
    let mut posts = Post::select_all(rb).await.unwrap_or_default();
    // 简单按 ID 倒序（假设 ID 自增）
    posts.sort_by(|a, b| b.id.cmp(&a.id));

    // 3. 生成 SUMMARY.md (左侧导航栏)
    let mut summary_content = String::from("# Summary\n\n[Introduction](README.md)\n\n# Blog Posts\n");
    
    // 4. 生成 README.md (首页)
    let readme_content = format!(
        "# Welcome to My Rust CMS\n\nCurrently there are {} posts generated by mdBook.", 
        posts.len()
    );
    fs::write(src_dir.join("README.md"), readme_content)?;

    // 5. 遍历文章，生成 Markdown 文件并写入 SUMMARY
    for post in posts {
        let file_name = format!("{}.md", post.slug);
        let file_path = src_dir.join(&file_name);

        // 写入文章内容
        // 可以在这里加个标题头
        let content_with_title = format!("# {}\n\n{}", post.title, post.content);
        fs::write(file_path, content_with_title)?;

        // 添加到目录
        // 格式: - [Title](filename.md)
        summary_content.push_str(&format!("- [{}]({})\n", post.title, file_name));
    }

    // 6. 写入 SUMMARY.md
    fs::write(src_dir.join("SUMMARY.md"), summary_content)?;

    // 7. 生成 book.toml (配置文件)
    let config_content = r#"
[book]
authors = ["Rust CMS"]
language = "en"
multilingual = false
src = "src"
title = "My CMS Blog"

[output.html]
additional-css = ["custom.css"]
"#;
    fs::write(Path::new(BOOK_ROOT).join("book.toml"), config_content)?;

    // 8. 调用 mdBook 执行构建
    // mdbook 构建是同步阻塞的 CPU 密集型任务，放到 blocking 线程跑
    tokio::task::spawn_blocking(move || {
        let md = MDBook::load(BOOK_ROOT).expect("Failed to load mdbook");
        md.build().expect("Failed to build mdbook");
    }).await?;

    println!("✅ mdBook build finished successfully!");
    Ok(())
}

// src/handlers/admin.rs
use axum::{
    extract::{Multipart, Query, State},
    response::{Html, IntoResponse, Json},
};
use std::sync::Arc;
use crate::{db::AppState, models::post::Post, book_gen::rebuild_book}; // 引入 rebuild_book
use serde_json::json;
use uuid::Uuid;
use chrono::Local;

// ... AuthParams, editor_view, upload_image 代码保持不变 ...

// 修改发布逻辑
pub async fn create_post(
    State(state): State<Arc<AppState>>,
    Json(payload): Json<serde_json::Value>,
) -> impl IntoResponse {
    let title = payload.get("title").unwrap().as_str().unwrap();
    let content = payload.get("content").unwrap().as_str().unwrap();
    
    let new_post = Post {
        id: None,
        title: title.to_string(),
        content: content.to_string(),
        slug: Uuid::new_v4().to_string(),
        created_at: Local::now().to_rfc3339(),
    };

    let _ = Post::insert(&state.rb, &new_post, "posts").await;

    // ----------------------------------------------------
    // [新增]：发布成功后，触发 mdBook 构建
    // ----------------------------------------------------
    if let Err(e) = rebuild_book(&state.rb).await {
        eprintln!("Failed to rebuild book: {}", e);
        return Json(json!({ "status": "error", "message": "Build failed" }));
    }

    // 返回成功，前端跳转地址改为生成的 HTML 页面地址
    // mdBook 生成的链接通常是 slug.html
    Json(json!({ "status": "ok", "slug": new_post.slug }))
}
4. 修改主程序 (src/main.rs)
现在的路由逻辑变了：
管理后台：依然走 Axum 动态路由。
前台博客：不再是动态渲染，而是直接把 mdBook 生成的 book/ 目录作为静态网站托管。
code

#[tokio::main]

    if let Err(e) = book_gen::rebuild_book(&rb).await {
        eprintln!("Warning: Initial book build failed: {}", e);
    }

    let book_output_dir = "data/book_workspace/book";

    let state = Arc::new(AppState { 
        rb,

        admin_token: admin_token.clone()

            .route("/api/upload", post(handlers::admin::upload_image))
            .layer(from_fn_with_state(state.clone(), middleware::auth_middleware))
        )
        // 兼容旧路由
        .route("/editor", get(handlers::admin::editor_view).layer(from_fn_with_state(state.clone(), middleware::auth_middleware)))
        .route("/api/posts", post(handlers::admin::create_post).layer(from_fn_with_state(state.clone(), middleware::auth_middleware)))

        .fallback_service(ServeDir::new(book_output_dir))
        
        .with_state(state);

    // ... (启动监听代码保持不变) ...
    let addr_str = format!("{}:{}", host, port);
    // ... 
    axum::serve(listener, app).await.unwrap();

6. 解决图片引用问题
mdBook 构建出来的静态网站位于 data/book_workspace/book。
而你上传的图片位于 static/uploads。
当 mdBook 页面渲染 <img src="/uploads/xxx.png"> 时：
浏览器请求 http://localhost:3000/uploads/xxx.png。
Axum 路由中配置了 .nest_service("/uploads", ...)。
完美！ 只要你在 Markdown 里写 /uploads/xxx.png，Axum 会拦截这个请求并返回图片，这与 mdBook 所在目录无关。
7. 删除无用代码 (清理)
既然改用了 mdBook 静态托管，那么原本的动态渲染逻辑就不需要了。
你可以删除或注释掉 src/handlers/blog.rs 以及相关的模板 templates/index.html, templates/post.html。系统现在的“首页”是 mdBook 生成的 index.html。
总结变更点
架构：Web Server (Axum) 变成了 "API Server + Static File Server"。
构建：引入 mdbook 库，并在 book_gen.rs 中实现了“数据库 -> 文件系统 -> HTML”的转换流。
触发：每次发布文章 (create_post) 都会重新构建整本书（对于个人博客来说，性能消耗可以忽略不计）。
展示：通过 .fallback_service(ServeDir::new(...)) 将生成的书作为网站根目录展示。
现在，你写完文章点击发布，后台会“卡顿”一下（正在构建 mdBook），然后跳转到生成的精美 mdBook 页面，左侧有目录导航，右侧是你的文章，支持搜索、夜间模式等 mdBook 所有原生功能。
